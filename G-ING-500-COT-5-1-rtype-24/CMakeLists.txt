cmake_minimum_required(VERSION 3.20)
project(r-type LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(RTYPE_BUILD_TESTS "Build unit tests" ON)
option(RTYPE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(RTYPE_BUILD_CLIENT "Build the graphical client (needs SFML)" ON)
option(RTYPE_BUILD_EXAMPLES "Build the example games made with the engine" ON)

list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
include(CompilerWarnings)
include(Dependencies)

if(RTYPE_BUILD_TESTS)
  enable_testing()
  include(GoogleTest)
endif()

add_subdirectory(engine)
add_subdirectory(network)
add_subdirectory(game)
add_subdirectory(server)
if(RTYPE_BUILD_CLIENT)
  add_subdirectory(client)
endif()
if(RTYPE_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
if(RTYPE_BUILD_TESTS)
  add_subdirectory(server/tests)
endif()
